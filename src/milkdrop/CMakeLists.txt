cmake_minimum_required(VERSION 3.21)
project(symphos_milkdrop LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(ENABLE_DEBUG_POSTFIX OFF CACHE BOOL "" FORCE)
add_subdirectory(../../vendor/projectm projectm)
target_sources(projectM PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/bridge.cpp)
target_include_directories(projectM PRIVATE ../../vendor/projectm/src/libprojectM ../../vendor/projectm/vendor/glad/include)
if(APPLE)
 target_compile_definitions(projectM PRIVATE GL_SILENCE_DEPRECATION)
 target_link_libraries(projectM PUBLIC "-framework OpenGL")
else()
 target_link_libraries(projectM PUBLIC EGL dl)
endif()
set_target_properties(projectM PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
